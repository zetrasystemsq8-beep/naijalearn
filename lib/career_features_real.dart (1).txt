/// REAL FEATURE 2: REAL ACHIEVEMENT SYSTEM (Track in AppProvider)
/// =========================================================================

class RealAchievementSystem {
  /// Call this after every quiz/battle to check achievements
  static Future<List<String>> checkAchievements(AppProvider provider) async {
    final unlockedNew = <String>[];
    final stats = provider.stats;
    final currentAchievements = stats.badges; // Use existing badges list

    // Achievement 1: First Quiz
    if (stats.quizzesCompleted >= 1 && !currentAchievements.contains('first_quiz')) {
      unlockedNew.add('first_quiz');
      await provider.addBadge('first_quiz');
    }

    // Achievement 2: 3-Day Streak
    if (stats.streak >= 3 && !currentAchievements.contains('streak_3')) {
      unlockedNew.add('streak_3');
      await provider.addBadge('streak_3');
      _showAchievementPopup('🔥 3-Day Streak!', 'You\'re on fire!');
    }

    // Achievement 3: 7-Day Streak
    if (stats.streak >= 7 && !currentAchievements.contains('streak_7')) {
      unlockedNew.add('streak_7');
      await provider.addBadge('streak_7');
      _showAchievementPopup('⚔️ Week Warrior!', '7-day streak - unstoppable!');
    }

    // Achievement 4: 80% Subject Master
    final subjects = provider.getAvailableSubjects();
    for (final subject in subjects) {
      if (provider.getSubjectScore(subject) >= 80 && 
          !currentAchievements.contains('subject_master_$subject')) {
        unlockedNew.add('subject_master_$subject');
        await provider.addBadge('subject_master_$subject');
      }
    }

    // Achievement 5: 100+ Questions
    if (stats.totalQuestionsAnswered >= 100 && 
        !currentAchievements.contains('centurion')) {
      unlockedNew.add('centurion');
      await provider.addBadge('centurion');
    }

    // Achievement 6: JAMB 180+
    final avgScore = provider.getAverageScore();
    final predictedJamb = ((avgScore / 100) * 240).round();
    if (predictedJamb >= 180 && !currentAchievements.contains('jamb_180')) {
      unlockedNew.add('jamb_180');
      await provider.addBadge('jamb_180');
    }

    return unlockedNew;
  }

  static void _showAchievementPopup(String title, String subtitle) {
    // Trigger notification/toast
    debugPrint('🏆 Achievement Unlocked: $title - $subtitle');
  }
}

/// =========================================================================
/// REAL FEATURE 3: REAL LEADERBOARD (From Supabase)
/// =========================================================================

class RealLeaderboardWidget extends StatefulWidget {
  const RealLeaderboardWidget({super.key});

  @override
  State<RealLeaderboardWidget> createState() => _RealLeaderboardWidgetState();
}

class _RealLeaderboardWidgetState extends State<RealLeaderboardWidget> {
  late Future<List<LeaderboardUser>> _leaderboardFuture;

  @override
  void initState() {
    super.initState();
    _leaderboardFuture = _fetchRealLeaderboard();
  }

  Future<List<LeaderboardUser>> _fetchRealLeaderboard() async {
    try {
      final supabase = Supabase.instance.client;
      
      // Fetch top 10 users by predicted JAMB score
      final response = await supabase
          .from('user_stats')
          .select('user_id, username, average_score, quiz_count, streak')
          .order('average_score', ascending: false)
          .limit(10);

      // Convert to LeaderboardUser objects
      final users = (response as List).map((data) {
        final predictedScore = ((data['average_score'] as num) / 100 * 240).round();
        return LeaderboardUser(
          userId: data['user_id'] as String,
          username: data['username'] as String ?? 'Anonymous',
          predictedJambScore: predictedScore,
          averageAccuracy: data['average_score'] as double,
          streak: data['streak'] as int? ?? 0,
        );
      }).toList();

      return users;
    } catch (e) {
      debugPrint('Error fetching leaderboard: $e');
      return [];
    }
  }

  @override
  Widget build(BuildContext context) {
    final scheme = Theme.of(context).colorScheme;

    return Container(
      margin: const EdgeInsets.all(16),
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: scheme.surfaceContainerHighest,
        borderRadius: BorderRadius.circular(20),
        border: Border.all(color: scheme.outlineVariant),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            '🏆 Top Scorers',
            style: TextStyle(
              fontSize: 16,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 12),
          FutureBuilder<List<LeaderboardUser>>(
            future: _leaderboardFuture,
            builder: (context, snapshot) {
              if (snapshot.connectionState == ConnectionState.waiting) {
                return Center(
                  child: CircularProgressIndicator(
                    strokeWidth: 2,
                    color: scheme.primary,
                  ),
                );
              }

              if (snapshot.hasError) {
                return Text('Error loading leaderboard: ${snapshot.error}');
              }

              final users = snapshot.data ?? [];
              if (users.isEmpty) {
                return const Text('No users yet');
              }

              return Column(
                children: users.asMap().entries.map((entry) {
                  final rank = entry.key + 1;
                  final user = entry.value;
                  final currentUser = Supabase.instance.client.auth.currentUser?.id;
                  final isMe = user.userId == currentUser;

                  return Container(
                    margin: const EdgeInsets.only(bottom: 8),
                    padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                    decoration: BoxDecoration(
                      color: isMe
                          ? scheme.primary.withOpacity(0.1)
                          : scheme.surfaceContainer,
                      borderRadius: BorderRadius.circular(12),
                      border: Border.all(
                        color: isMe
                            ? scheme.primary.withOpacity(0.3)
                            : Colors.transparent,
                      ),
                    ),
                    child: Row(
                      children: [
                        Text(
                          _getRankEmoji(rank),
                          style: const TextStyle(fontSize: 16),
                        ),
                        const SizedBox(width: 10),
                        Expanded(
                          child: Column(
                            crossAxisAlignment: CrossAxisAlignment.start,
                            children: [
                              Text(
                                isMe ? '${user.username} (You)' : user.username,
                                style: TextStyle(
                                  fontWeight: isMe
                                      ? FontWeight.bold
                                      : FontWeight.normal,
                                  fontSize: 13,
                                ),
                              ),
                              Text(
                                '${user.averageAccuracy.toStringAsFixed(0)}% • 🔥${user.streak}d',
                                style: TextStyle(
                                  fontSize: 11,
                                  color: scheme.onSurfaceVariant,
                                ),
                              ),
                            ],
                          ),
                        ),
                        Container(
                          padding: const EdgeInsets.symmetric(
                              horizontal: 10, vertical: 4),
                          decoration: BoxDecoration(
                            color: Colors.blue.withOpacity(0.15),
                            borderRadius: BorderRadius.circular(8),
                          ),
                          child: Text(
                            '${user.predictedJambScore}/240',
                            style: const TextStyle(
                              fontWeight: FontWeight.bold,
                              fontSize: 12,
                              color: Colors.blue,
                            ),
                          ),
                        ),
                      ],
                    ),
                  );
                }).toList(),
              );
            },
          ),
        ],
      ),
    );
  }

  String _getRankEmoji(int rank) {
    switch (rank) {
      case 1:
        return '👑';
      case 2:
        return '🥈';
      case 3:
        return '🥉';
      default:
        return '#$rank';
    }
  }
}

class LeaderboardUser {
  final String userId;
  final String username;
  final int predictedJambScore;
  final double averageAccuracy;
  final int streak;

  LeaderboardUser({
    required this.userId,
    required this.username,
    required this.predictedJambScore,
    required this.averageAccuracy,
    required this.streak,
  });
}

class RealSmartRecommendationsWidget extends StatelessWidget {
  const RealSmartRecommendationsWidget({super.key});

  @override
  Widget build(BuildContext context) {
    final provider = context.watch<AppProvider>();
    final scheme = Theme.of(context).colorScheme;

    // Find subjects 50-75% accuracy (sweet spot for improvement)
    final subjects = provider.getAvailableSubjects();
    final recommendations = <String, double>{};

    for (final subject in subjects) {
      final score = provider.getSubjectScore(subject);
      // Focus on improvement zone
      if (score >= 50 && score < 75) {
        recommendations[subject] = score;
      }
    }

    // If all subjects are strong, suggest 75+ subjects
    if (recommendations.isEmpty) {
      for (final subject in subjects) {
        final score = provider.getSubjectScore(subject);
        if (score >= 75 && score < 90) {
          recommendations[subject] = score;
        }
      }
    }

    if (recommendations.isEmpty) {
      return const SizedBox.shrink();
    }

    final sorted = recommendations.entries.toList()
      ..sort((a, b) => a.value.compareTo(b.value));

    return Container(
      margin: const EdgeInsets.all(16),
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: scheme.surfaceContainerHighest,
        borderRadius: BorderRadius.circular(20),
        border: Border.all(color: scheme.outlineVariant),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            '💡 Recommended Next',
            style: TextStyle(
              fontSize: 16,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 12),
          ...sorted.take(3).map((entry) {
            final targetScore = 75;
            final gapToTarget = (targetScore - entry.value).round();
            final timeEstimate = (gapToTarget * 1.5).toInt();

            return GestureDetector(
              onTap: () {
                final subjectInfo = kSubjects.firstWhere((s) => s.name == entry.key);
                final set = QuestionRepository.getForSubject(entry.key)..shuffle();
                Navigator.of(context).push(MaterialPageRoute(
                  builder: (_) => ExamInstructionsScreen(
                    subject: subjectInfo,
                    questions: set.take(20).toList(),
                  ),
                ));
              },
              child: Container(
                margin: const EdgeInsets.only(bottom: 10),
                padding: const EdgeInsets.all(12),
                decoration: BoxDecoration(
                  color: scheme.surfaceContainer,
                  borderRadius: BorderRadius.circular(14),
                  border: Border.all(color: scheme.outlineVariant.withOpacity(0.5)),
                ),
                child: Row(
                  children: [
                    Text(
                      '📚',
                      style: const TextStyle(fontSize: 20),
                    ),
                    const SizedBox(width: 10),
                    Expanded(
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Text(
                            entry.key,
                            style: const TextStyle(
                              fontWeight: FontWeight.bold,
                              fontSize: 13,
                            ),
                          ),
                          Text(
                            '${entry.value.toStringAsFixed(0)}% → Need +$gapToTarget to hit 75% (~${timeEstimate}min)',
                            style: TextStyle(
                              fontSize: 11,
                              color: scheme.onSurfaceVariant,
                            ),
                          ),
                        ],
                      ),
                    ),
                    Icon(
                      Icons.arrow_forward_rounded,
                      size: 18,
                      color: scheme.primary,
                    ),
                  ],
                ),
              ),
            );
          }).toList(),
        ],
      ),
    );
  }
}
